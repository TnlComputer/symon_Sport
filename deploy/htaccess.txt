# Reglas para el .htaccess del hosting (Apache).
# NO reemplazar el .htaccess actual del servidor: AGREGAR estas reglas debajo de las del SSL.

# Página de inicio
DirectoryIndex index.php

<IfModule mod_rewrite.c>
  RewriteEngine On

  # Carpetas internas: no se pueden abrir desde el navegador
  RewriteRule ^(includes|sql|deploy|php/connections)(/|$) - [F,L]
  RewriteRule ^router\.php$ - [F,L]
  # Archivos y carpetas ocultos (.git, .env, .vscode...), salvo .well-known (certificado SSL)
  RewriteRule (^|/)\.(?!well-known/) - [F,L]

  # Direcciones viejas .html -> página nueva .php (301: Google actualiza sus enlaces)
  RewriteCond %{DOCUMENT_ROOT}/$1.php -f
  RewriteRule ^(.+)\.html$ /$1.php [R=301,L,QSA]
</IfModule>

# No listar el contenido de las carpetas
Options -Indexes

# Compresión
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript application/json image/svg+xml
</IfModule>

# Caché de archivos estáticos (CSS/JS llevan ?v=..., así que se actualizan al cambiar la versión)
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/png "access plus 1 month"
  ExpiresByType image/webp "access plus 1 month"
  ExpiresByType image/svg+xml "access plus 1 month"
  ExpiresByType image/x-icon "access plus 1 month"
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
</IfModule>

# Archivos sensibles
<FilesMatch "\.(sql|md|txt|log|zip)$">
  Require all denied
</FilesMatch>
